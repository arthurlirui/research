


	
\section{Information Entropy Optimization Objective Functions}

\subsection{Maximum Coverage Information Entropy Objective}
\begin{equation}
	\max \sum_{i=1}^N H(v_i) \cdot \mathbb{I}(v_i \in \mathcal{V}_{\text{visible}})
\end{equation}
where:
\begin{itemize}
	\item $H(v_i) = -\sum_{k=1}^K p_k(v_i)\log p_k(v_i)$ represents the information entropy of voxel $v_i$
	\item $p_k(v_i)$ is the probability of voxel belonging to semantic class $k$ (output by VLLM)
	\item $\mathbb{I}(v_i \in \mathcal{V}_{\text{visible}})$ is the visibility indicator function
	\item $N$ is the total number of discretized voxels in the scene
\end{itemize}

\subsection{Value-Weighted Coverage Objective}
\begin{equation}
	\max \sum_{j \in \mathcal{R}} w_j \cdot \text{QoD}(p_j)
\end{equation}
where:
\begin{itemize}
	\item $\mathcal{R}$ is the set of regions of interest (ROI)
	\item $w_j = \frac{e^{s_j}}{\sum e^{s_k}}$ is the semantic importance weight from VLLM
	\item $\text{QoD}(p_j) = \|\nabla D(p_j)\|_2 \cdot \text{SSIM}(p_j)$ is the quality-of-data metric for point $p_j$
\end{itemize}

\subsection{Multi-Camera Coordination Constraints}
\begin{equation}
	\begin{aligned}
		\text{s.t.} \quad & \bigcap_{m=1}^M \mathcal{FOV}_m \geq \emptyset \\
		& \|\mathbf{T}_m - \mathbf{T}_n\|_2 \geq d_{\text{min}}, \quad \forall m \neq n
	\end{aligned}
\end{equation}
where:
\begin{itemize}
	\item $\mathcal{FOV}_m$ is the view frustum of the $m$-th camera
	\item $\mathbf{T}_m$ is the camera pose (SE(3) transformation matrix)
	\item $d_{\text{min}}$ is the minimum safety distance
\end{itemize}

\subsection{Dynamic Reconstruction Objective}
For time-varying scenes:
\begin{equation}
	\max_{\{\theta_t\}} \sum_{\tau=t}^{t+H} \gamma^{t-\tau} \mathbb{E}[H_{\tau}(v|\theta_\tau)]
\end{equation}
where:
\begin{itemize}
	\item $\theta_t$ is the camera pose parameter at time $t$
	\item $H$ is the prediction horizon
	\item $\gamma$ is the discount factor
	\item $H_{\tau}(v|\theta_\tau)$ is the conditional information entropy
\end{itemize}

\subsection{Formula Notes}
\begin{enumerate}
	\item \textbf{Information entropy calculation} uses sliding window update:
	\begin{equation}
		H_t(v_i) = \alpha H_{t-1}(v_i) + (1-\alpha)\hat{H}_t(v_i)
	\end{equation}
	where $\alpha=0.7$ is the forgetting factor
	
	\item \textbf{Semantic weights} are generated by VLLM's attention mechanism:
	\begin{equation}
		w_j = \text{softmax}(\text{Attn}(Q_j, K_{\mathcal{R}}))
	\end{equation}
\end{enumerate}
